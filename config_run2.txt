import yaml, torch
from pathlib import Path

data_root = Path(DATA_ROOT)

cfg_path = Path(WORKSPACE) / 'firest_graphnet' / 'configs' / 'base.yaml'
with open(cfg_path) as f:
    cfg = yaml.safe_load(f)

cfg['data']['root'] = str(data_root)
cfg['forecast']['enabled'] = True
cfg['device'] = 'cuda' if torch.cuda.is_available() else 'cpu'

try:
    cfg['forecast']['T_in']    = RECOMMENDED_T_IN
    cfg['forecast']['horizon'] = RECOMMENDED_HORIZON
except NameError:
    pass  # keep yaml defaults

cfg['training']['epochs']          = 80
cfg['training']['batch_size']      = 4
cfg['training']['lr']              = 5e-4    # was 1e-3
cfg['training']['weight_decay']    = 5e-4    # was 1e-4
cfg['training']['pos_weight']      = 5.0     # was 10.0
cfg['training']['patience']        = 5       # early stop after 5 val checks
cfg['training']['mixed_precision'] = True
cfg['training']['val_every']       = 2
cfg['training']['metric_every']    = 20
cfg['pinn']['beta_max']            = 0.05    # was 0.01
cfg['pinn']['eta_max']             = 0.2     # was 0.1

DEVICE = torch.device(cfg['device'])
fcfg = cfg['forecast']
dcfg = cfg['data']
print(f"Config loaded | Device: {DEVICE}")
print(f"lr={cfg['training']['lr']} | pos_weight={cfg['training']['pos_weight']} | weight_decay={cfg['training']['weight_decay']}")
print(f"epochs={cfg['training']['epochs']} | patience={cfg['training']['patience']} val-checks")
